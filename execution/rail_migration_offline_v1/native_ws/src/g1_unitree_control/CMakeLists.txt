cmake_minimum_required(VERSION 3.5)
project(g1_unitree_control)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(orocos_kdl REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(control_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(unitree_api REQUIRED)
find_package(unitree_hg REQUIRED)
find_package(urdf REQUIRED)
find_package(kdl_parser REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "srv/ResumeTaskHold.srv")
add_executable(arm_sdk_hardware_gateway src/arm_sdk_hardware_gateway.cpp)
target_compile_features(arm_sdk_hardware_gateway PRIVATE cxx_std_17)
ament_target_dependencies(arm_sdk_hardware_gateway control_msgs diagnostic_msgs rcl_interfaces rclcpp rclcpp_action sensor_msgs std_srvs trajectory_msgs unitree_api unitree_hg urdf kdl_parser)
target_link_libraries(arm_sdk_hardware_gateway Eigen3::Eigen orocos-kdl)
install(TARGETS arm_sdk_hardware_gateway DESTINATION lib/${PROJECT_NAME})
add_executable(arm_state_bridge src/arm_state_bridge.cpp)
target_compile_features(arm_state_bridge PRIVATE cxx_std_17)
ament_target_dependencies(arm_state_bridge control_msgs diagnostic_msgs rcl_interfaces rclcpp rclcpp_action sensor_msgs std_srvs trajectory_msgs unitree_api unitree_hg urdf kdl_parser)
target_link_libraries(arm_state_bridge Eigen3::Eigen orocos-kdl)
install(TARGETS arm_state_bridge DESTINATION lib/${PROJECT_NAME})
target_compile_options(arm_sdk_hardware_gateway PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/include/foxy_compat.hpp)
rosidl_target_interfaces(arm_sdk_hardware_gateway ${PROJECT_NAME} "rosidl_typesupport_cpp")
ament_package()
