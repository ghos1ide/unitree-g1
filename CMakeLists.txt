function(rh56_require_files component)
  set(missing)
  foreach(file IN LISTS ARGN)
    set(path "${PROJECT_SOURCE_DIR}/${file}")
    if(EXISTS "${path}" AND NOT IS_DIRECTORY "${path}")
      file(READ "${path}" first_byte LIMIT 1 HEX)
      if(NOT first_byte STREQUAL "")
        continue()
      endif()
    endif()
    list(APPEND missing "${file}")
  endforeach()
  if(missing)
    string(REPLACE ";" "\n  " missing "${missing}")
    message(FATAL_ERROR "${component} needs these missing or empty files:\n  ${missing}\nSee README.md for migration details.")
  endif()
endfunction()

  endfunction()
